function secondHighest(arr) { if (arr.length <= 1) { return -Infinity; } let allSame = true; for (let i = 1; i < arr.length; i++) { if (arr[i] !== arr[0]) { allSame = false; break; } } if (allSame) { return -Infinity; } let highest = -Infinity; let second = -Infinity; for (let num of arr) { if (num > highest) { second = highest; highest = num; } else if (num > second && num < highest) { second = num; } } return second; }
